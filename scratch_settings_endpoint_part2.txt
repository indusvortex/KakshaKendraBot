3923:   if (!text) return null;
3924:   text = String(text).toLowerCase().trim();
3925:   var now = new Date();
3926:   var d = new Date(now.getTime());
3927: 
3928:   // Day modifiers
3929:   if (text.indexOf("parso") !== -1 || text.indexOf("day after") !== -1) {{
3930:     d.setDate(d.getDate() + 2);
3931:   }} else if (text.indexOf("kal") !== -1 || text.indexOf("tomorrow") !== -1) {{
3932:     d.setDate(d.getDate() + 1);
3933:   }} else if (text.indexOf("aaj") !== -1 || text.indexOf("today") !== -1) {{
3934:     /* today */
3935:   }}
3936: 
3937:   // Time match: 5pm / 5 PM / 5:30 pm / 17:00 / 5
3938:   var m = text.match(/(\\d{{1,2}})(?:[:.](\\d{{2}}))?\\s*(am|pm)?/);
3939:   if (!m) {{ d.setHours(12, 0, 0, 0); return d; }}
3940:   var h = parseInt(m[1], 10);
3941:   var mm = parseInt(m[2] || "0", 10);
3942:   var ampm = m[3];
3943:   if (ampm === "pm" && h < 12) h += 12;
3944:   if (ampm === "am" && h === 12) h = 0;

3946:   if (!ampm && h >= 1 && h <= 7) h += 12;
